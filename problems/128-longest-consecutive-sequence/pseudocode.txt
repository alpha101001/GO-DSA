PROBLEM: 128. Longest Consecutive Sequence
LINK: https://leetcode.com/problems/longest-consecutive-sequence/description/
FUNCTION: func longestConsecutive(nums []int) int

----------------------------------
--------------CODE----------------
----------------------------------
set <- all numbers
best <-1

for val in range set:
  if val-1 in set: continue
  length <-1
  while val+length in set:
      length++;
  best <- max(best,length)
