PROBLEM: 238. Product of Array Except Self
LINK: https://leetcode.com/problems/product-of-array-except-self/description/
FUNCTION: func productExceptSelf(nums []int) []int

----------------------------------
--------------CODE----------------
----------------------------------
given arr[]

n = len(arr)
prefix=1
res = make([]int,n)
for i<- 0 to n-1:
    res[i] <- prefix
    prefix <- prefix*arr[i]
suffix =1
for i <- n-1 down to 0:
  res[i] <- res[i]*suffix
  suffix <- suffix * arr[i]

return res
