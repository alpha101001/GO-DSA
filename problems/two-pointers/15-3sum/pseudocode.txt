PROBLEM: 15. 3Sum
LINK: https://leetcode.com/problems/3sum/description/
FUNCTION: func threeSum(nums []int) [][]int

----------------------------------
--------------CODE----------------
----------------------------------
n <- len(arr)
sort arr
i <- 0

for i <- 0 to n-3:
if i>0 && arr[i] == arr[i-1] : continue
  left <- i+1
  right <- n-1
  while left<right:
    sum = arr[i]+arr[left]+arr[right]
    if sum == 0 record left++ right--;
    else if sum<0 : left++;
    else if sum>0 : right--
