sort(arr)
n <- len(arr)
count <-0
for k <- n-1 down to 2:
  left <- 0
  right <- k-1
  while left<right:
    valid = arr[left]+arr[right] > arr[k]
    if valid :
      count = count+ (right-left)
      right--;
    else:
      left++
